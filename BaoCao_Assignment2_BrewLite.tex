\section{TASK 1: KHAI THÁC VÀ ĐẶC TẢ YÊU CẦU}

\subsection{Task 1.1 -- Bối cảnh, Stakeholder và Phạm vi Hệ thống}

Tài liệu tham chiếu bắt buộc: \texttt{docs/requirements/context-and-scope.md}.

\subsubsection{Vấn đề nghiệp vụ cần giải quyết}
Trong mô hình kinh doanh quán cà phê và đồ uống truyền thống, các cửa hàng thường xuyên gặp phải những bất cập sau trong khâu vận hành và phục vụ khách hàng:
\begin{enumerate}[leftmargin=*]
    \item \textbf{Quá tải và chờ đợi lâu vào giờ cao điểm:} Khách hàng phải xếp hàng dài để chờ nhân viên ghi nhận món tại quầy, gây gián đoạn và giảm tỷ lệ hài lòng của khách.
    \item \textbf{Sai sót trong tiếp nhận và pha chế order:} Khách hàng có nhu cầu tùy biến kích cỡ (Size S/M/L) và chọn nhiều topping đi kèm. Việc truyền đạt trực tiếp bằng miệng hoặc ghi chép tay dễ dẫn đến nhầm lẫn, làm sai công thức gây lãng phí nguyên liệu.
    \item \textbf{Thiếu tính minh bạch trong theo dõi tiến độ đơn hàng:} Khách hàng không nắm được đơn của mình đang ở giai đoạn nào (chờ thanh toán, đang pha chế, hay đã xong để lấy), dẫn đến việc tụ tập chờ đợi quanh quầy bar gây lộn xộn.
    \item \textbf{Quản lý tồn kho và doanh thu phân tán:} Khó kiểm soát tồn kho tức thời (real-time stock) dẫn đến tình trạng khách đặt món nhưng thực tế nguyên liệu đã hết; thiếu cơ chế đối soát thanh toán và rủi ro đặt trùng đơn do lỗi mạng.
    \item \textbf{Thiếu cơ chế gắn kết khách hàng (Customer Loyalty):} Cửa hàng chưa có giải pháp tự động tích lũy điểm thưởng thành viên theo giá trị đơn hàng để tái thu hút khách hàng quay lại.
\end{enumerate}

\noindent\textbf{Giải pháp của BrewLite:} Hệ thống xây dựng nền tảng đặt đồ uống trực tuyến gọn nhẹ, tối ưu cho thiết bị di động (Mobile-First / PWA), kết nối trực tiếp và đồng bộ theo thời gian thực giữa Khách hàng, Nhân viên quầy bar (Barista) và Quản lý cửa hàng.

\subsubsection{Phạm vi hệ thống (In-Scope và Out-of-Scope)}

\noindent\textbf{A. Phạm vi trong hệ thống (In-Scope):}
\begin{itemize}[leftmargin=*,noitemsep]
    \item \textbf{Phạm vi MVP Sprint 1:} Xem danh mục thực đơn đồ uống theo danh mục (Cà phê, Trà, Trà sữa, Khác); Xem chi tiết sản phẩm; Giao diện chuẩn Mobile-first Responsive; Xử lý 3 trạng thái bắt buộc (Loading Skeleton, Empty State, Error State); Kịch bản database seed dữ liệu mẫu; Kiểm thử tự động cơ bản.
    \item \textbf{Phạm vi Sprint 2 (Hoàn thiện MVP):} Tùy biến kích cỡ ly (Size S, M, L) và chọn Topping kèm theo; Giỏ hàng quản lý bằng Zustand kết hợp \texttt{localStorage}; Đăng ký và đăng nhập bảo mật bằng mật khẩu băm bcrypt và mã JWT; Tạo đơn hàng với giao dịch cơ sở dữ liệu atomic; Thanh toán không tiền mặt (mô phỏng VietQR / MoMo) với khóa chống trùng \texttt{idempotencyKey}; Theo dõi trạng thái đơn hàng thời gian thực qua máy trạng thái 7 bước; Phân hệ Barista KDS điều phối đơn; Tích lũy điểm thưởng thành viên tự động và áp dụng chiết khấu theo thứ hạng (Đồng, Bạc, Vàng, Kim Cương).
\end{itemize}

\noindent\textbf{B. Phạm vi ngoài hệ thống (Out-of-Scope):}
\begin{itemize}[leftmargin=*,noitemsep]
    \item Không tích hợp cổng thanh toán thương mại live thực tế (VNPay, Stripe) vì yêu cầu đồ án chỉ sử dụng dịch vụ thanh toán mô phỏng an toàn.
    \item Không tích hợp định vị GPS hoặc đơn vị vận chuyển đường dài ngoài (GrabExpress, ShopeeFood). Khách hàng nhận món trực tiếp tại quầy quán (Pick-up at counter).
    \item Không tích hợp phần cứng máy in bill nhiệt ESC/POS hoặc ngăn kéo đựng tiền vật lý.
    \item Không phát triển phân hệ chấm công, xếp lịch ca làm việc và tính lương nhân viên.
\end{itemize}

\subsubsection{Giả định, Ràng buộc kỹ thuật \& Đánh giá rủi ro ban đầu}

\noindent\textbf{A. Giả định (Assumptions):} Khách hàng sử dụng thiết bị có trình duyệt web hiện đại hỗ trợ JavaScript/HTML5 và kết nối mạng Internet ổn định. Quy mô phục vụ một cửa hàng đơn lẻ với lưu lượng dưới 100 giao dịch đồng thời trong giờ cao điểm. Khách hàng luôn có đủ số dư tài khoản khi thanh toán thử nghiệm.

\noindent\textbf{B. Ràng buộc kỹ thuật bắt buộc:}
\begin{itemize}[leftmargin=*,noitemsep]
    \item \textbf{Frontend:} Bắt buộc sử dụng Next.js (App Router) theo định hướng Mobile-first / PWA.
    \item \textbf{Backend:} Bắt buộc sử dụng NestJS theo kiến trúc module/domain.
    \item \textbf{Database:} Bắt buộc sử dụng PostgreSQL 16 kết hợp Prisma ORM và migration có phiên bản.
    \item \textbf{Containerization:} Bắt buộc đóng gói và chạy được toàn bộ hệ thống bằng \texttt{docker-compose.yml}.
    \item \textbf{Toàn vẹn dữ liệu:} Thao tác tạo đơn hàng và trừ tồn kho nguyên liệu phải nằm trong một Database Transaction (Atomic); Chống race condition bán âm kho bằng trường \texttt{version} (Optimistic Locking).
\end{itemize}

\subsubsection{Kịch bản thực thi chi tiết (Flow of Events)}

\noindent\textbf{1. Luồng sự kiện chính (Main Flow / Happy Path):}
\begin{enumerate}[leftmargin=*,noitemsep]
    \item Khách hàng mở trang Giỏ hàng (\texttt{/cart}) hoặc trang Thanh toán (\texttt{/checkout}), kiểm tra danh sách món, kích cỡ (S/M/L) và topping.
    \item Hệ thống tự động truy vấn số điểm tích lũy \texttt{loyaltyPoints} của khách hàng qua JWT và tính mức chiết khấu thành viên tương ứng (0\%, 5\%, 10\% hoặc 15\%).
    \item Khách hàng chọn phương thức thanh toán không tiền mặt (Ví MoMo hoặc Quét mã VietQR).
    \item Khách hàng nhấn nút xác nhận đặt hàng và thanh toán. Client sinh chuỗi định danh ngẫu nhiên UUID làm \texttt{idempotencyKey}.
    \item Next.js gửi yêu cầu \texttt{POST /api/orders} kèm JWT Bearer token và danh sách món trong giỏ hàng.
    \item Backend NestJS mở một Prisma Transaction:
    \begin{itemize}[leftmargin=*,noitemsep]
        \item Xác thực phiên đăng nhập người dùng qua \texttt{JwtAuthGuard}.
        \item Khóa dữ liệu kiểm tra tồn kho nguyên liệu cần thiết cho từng ly nước theo công thức (BOM).
        \item Kiểm tra trường \texttt{version} của sản phẩm (Optimistic Concurrency Control).
        \item Khấu trừ tồn kho nguyên vật liệu (\texttt{Ingredient.stock}) và tạo đơn hàng mới với trạng thái \texttt{PENDING}.
    \end{itemize}
    \item Client gửi yêu cầu thực hiện thanh toán \texttt{POST /api/payments} kèm \texttt{orderId}, số tiền và \texttt{idempotencyKey}.
    \item Backend kiểm tra \texttt{idempotencyKey}. Do đây là lần gửi đầu tiên, hệ thống ghi nhận giao dịch thanh toán thành công, cập nhật trạng thái đơn hàng sang \texttt{PAID}, cộng điểm thưởng thành viên vào bảng \texttt{LoyaltyTransaction}, hoàn tất transaction và trả về mã phản hồi HTTP \texttt{200 OK}.
    \item Giao diện Next.js xóa sạch giỏ hàng trong Zustand store và chuyển hướng khách hàng sang trang Xác nhận đơn hàng thành công (\texttt{/orders/:id}).
\end{enumerate}

\vspace{0.2cm}
\noindent\textbf{2. Các luồng sự kiện thay thế (Alternative Flows):}
\begin{itemize}[leftmargin=*,noitemsep]
    \item \textbf{A1 -- Áp dụng mã Voucher khuyến mãi bổ sung:} Tại bước 2, khách hàng nhập mã ưu đãi (ví dụ: \texttt{COFFEELOVER}). Backend kiểm tra tính hợp lệ trong bảng \texttt{Promotion} (còn hạn, đạt giá trị đơn tối thiểu), áp dụng giảm giá trên tổng tiền trước khi tạo đơn.
    \item \textbf{A2 -- Khách hàng thay đổi phương thức thanh toán:} Tại bước 3, khách hàng chuyển đổi linh hoạt giữa quét mã VietQR và ví MoMo. Giao diện hiển thị mã QR động tương ứng từ dịch vụ thanh toán mà không làm thay đổi nội dung giỏ hàng.
\end{itemize}

\vspace{0.2cm}
\noindent\textbf{3. Các luồng sự kiện ngoại lệ (Exception Flows):}
\begin{itemize}[leftmargin=*,noitemsep]
    \item \textbf{E1 -- Hết nguyên liệu tồn kho hoặc xung đột phiên bản (Race Condition):} Tại bước 6, nếu nguyên liệu pha chế không đủ hoặc phiên bản \texttt{Product.version} bị lệch do khách hàng khác vừa đặt trước, Backend lập tức hủy bỏ transaction (\texttt{ROLLBACK}), trả về lỗi HTTP \texttt{400 Bad Request} kèm thông báo: \textit{``Sản phẩm hoặc nguyên liệu đã hết hàng, vui lòng điều chỉnh giỏ hàng''}. Không có đơn hàng dở dang nào được lưu lại.
    \item \textbf{E2 -- Hết hạn phiên đăng nhập (JWT Expired):} Tại bước 5, nếu token JWT hết hạn, Backend từ chối với mã HTTP \texttt{401 Unauthorized}. Giao diện lưu tạm giỏ hàng vào \texttt{localStorage}, mở popup thông báo và chuyển hướng khách sang trang Đăng nhập (\texttt{/login}). Sau khi đăng nhập lại, khách được đưa trở về trang checkout để tiếp tục.
    \item \textbf{E3 -- Thanh toán thất bại hoặc người dùng hủy giao dịch:} Tại bước 8, nếu giao dịch thanh toán bị từ chối (tài khoản không đủ số dư mô phỏng hoặc lỗi kết nối), đơn hàng được chuyển sang trạng thái \texttt{PAYMENT\_FAILED}. Hệ thống kích hoạt quy trình hoàn trả tồn kho nguyên liệu đã giữ chỗ, giỏ hàng tại Client được bảo lưu nguyên vẹn và hiển thị nút \textit{``Thử thanh toán lại''}.
    \item \textbf{E4 -- Khách hàng nhấn thanh toán nhiều lần hoặc gửi lại request (Idempotency Hit):} Nếu do mạng chập chờn mà Client gửi lại request \texttt{POST /api/payments} cùng \texttt{idempotencyKey}, Backend kiểm tra bảng \texttt{Payment}, phát hiện key đã tồn tại và lập tức trả về kết quả thành công của giao dịch trước đó mà không thực hiện trừ tiền hay cộng điểm lần 2.
\end{itemize}

    \item \textbf{Nhánh kiểm tra tồn kho nguyên liệu:} Khi tạo đơn, Backend tính toán định mức tiêu hao nguyên vật liệu theo kích cỡ ly. Nếu phát hiện nguyên liệu bị thiếu hụt, hệ thống rẽ sang nhánh thất bại, rollback transaction và thông báo lỗi rõ ràng cho người dùng.

    \end{itemize}
\end{itemize}

\subsubsection{Phân tích các giai đoạn tương tác tuần tự}

\begin{enumerate}[leftmargin=*]
    \item \textbf{Giai đoạn 1 -- Khởi tạo phiên \& Xác thực JWT:}
    \begin{itemize}[leftmargin=*,noitemsep]
        \item Client gửi HTTP Request: \texttt{POST /api/orders} kèm header \texttt{Authorization: Bearer <JWT>} và payload chứa danh sách món trong giỏ hàng.
        \item \texttt{JwtAuthGuard} tại Backend giải mã chữ ký token bằng \texttt{JWT\_SECRET}. Nếu token không hợp lệ hoặc hết hạn, trả về ngay HTTP \texttt{401 Unauthorized}.
    \end{itemize}
    \item \textbf{Giai đoạn 2 -- Mở Database Transaction \& Khóa tồn kho nguyên liệu:}
    \begin{itemize}[leftmargin=*,noitemsep]
        \item Backend mở khối giao dịch \texttt{prisma.\$transaction(async (tx) => \{...\})}.
        \item Truy vấn danh mục định mức nguyên liệu (\texttt{ProductRecipeItem}) của từng sản phẩm theo kích cỡ được chọn.
        \item Kiểm tra trường \texttt{version} của sản phẩm để chống hiện tượng Race Condition khi nhiều khách hàng cùng mua ly nước cuối cùng.
        \item Cập nhật khấu trừ số lượng nguyên liệu: \texttt{UPDATE "Ingredient" SET stock = stock - amount}.
        \item Khởi tạo bản ghi \texttt{Order} ở trạng thái ban đầu là \texttt{PENDING} kèm danh sách chi tiết các dòng món \texttt{OrderItem} và toppings đính kèm.
        \item Phản hồi mã HTTP \texttt{201 Created} kèm thông tin đơn hàng vừa khởi tạo về cho Next.js UI.
    \end{itemize}
    \item \textbf{Giai đoạn 3 -- Xử lý giao dịch thanh toán \& Cơ chế Idempotency Key:}
    \begin{itemize}[leftmargin=*,noitemsep]
        \item Next.js UI gửi tiếp yêu cầu thanh toán: \texttt{POST /api/payments} kèm \texttt{orderId}, số tiền, phương thức (\texttt{WALLET} / \texttt{CARD}) và chuỗi định danh duy nhất \texttt{idempotencyKey} (UUID v4).
        \item Backend kiểm tra sự tồn tại của \texttt{idempotencyKey} trong bảng \texttt{Payment}:
        \begin{itemize}[leftmargin=*,noitemsep]
            \item \textbf{Nhánh xử lý lặp (Idempotent Hit):} Nếu key đã được xử lý trước đó, hệ thống lập tức hoàn trả dữ liệu phản hồi đã lưu, không thực hiện trừ tiền lại hay thay đổi trạng thái đơn hàng.
            \item \textbf{Nhánh xử lý mới (New Request):} Backend chuyển tiếp yêu cầu đến Dịch vụ thanh toán mô phỏng.
        \end{itemize}
    \end{itemize}
    \item \textbf{Giai đoạn 4 -- Phân nhánh kết quả thanh toán (Nhánh Alt):}
    \begin{itemize}[leftmargin=*,noitemsep]
        \item \textbf{[Thanh toán Thành công]:} Dịch vụ thanh toán phản hồi \texttt{SUCCESS}. Backend cập nhật trạng thái đơn hàng sang \texttt{PAID}, tạo bản ghi giao dịch trong bảng \texttt{Payment}, tính toán điểm thưởng và ghi nhận vào bảng \texttt{LoyaltyTransaction}, hoàn tất transaction và trả về HTTP \texttt{200 OK}. Giao diện Client dọn sạch giỏ hàng và chuyển sang trang \texttt{/orders/:id}.
        \item \textbf{[Thanh toán Thất bại]:} Dịch vụ thanh toán phản hồi \texttt{FAILED}. Backend cập nhật đơn hàng thành \texttt{PAYMENT\_FAILED}, kích hoạt hoàn trả số lượng nguyên liệu đã trừ, trả về lỗi HTTP \texttt{400 Bad Request}. Giao diện Client giữ nguyên giỏ hàng và hiển thị thông báo lỗi kèm nút cho phép thử lại an toàn.
    \end{itemize}
\end{enumerate}

\subsubsection{Mô hình C4 (The C4 Model for Software Architecture)}

Kiến trúc hệ thống BrewLite được phân rã theo chuẩn \textbf{C4 Model} nhằm thể hiện rõ ràng các tầng trừu tượng từ bức tranh tổng thể đến cấu trúc module chi tiết bên trong mã nguồn:

\begin{enumerate}[leftmargin=*]
    \item \textbf{C4 Level 1 -- System Context (Ngữ cảnh hệ thống):}
    \begin{itemize}[leftmargin=*,noitemsep]
        \item \textit{Khách hàng (Customer):} Tương tác trực tiếp với BrewLite qua trình duyệt điện thoại để xem menu, đặt đồ uống và thanh toán.
        \item \textit{Nhân viên quầy (Barista):} Tương tác với BrewLite qua màn hình quầy bar (KDS) để nhận đơn và báo trạng thái pha chế.
        \item \textit{Hệ thống BrewLite:} Cung cấp toàn bộ dịch vụ cốt lõi quản lý menu, đơn hàng và tài khoản.
        \item \textit{Mock Payment Service:} Hệ thống bên ngoài mô phỏng cổng thanh toán Ví/Thẻ, trao đổi qua giao thức HTTP an toàn với cơ chế Idempotency Key.
    \end{itemize}
    
    \item \textbf{C4 Level 2 -- Container (Các khối chứa độc lập):}
    \begin{itemize}[leftmargin=*,noitemsep]
        \item \textbf{Web Application (Frontend PWA):} Ứng dụng Next.js 16 (React 19, Tailwind CSS 4, Zustand) phục vụ giao diện người dùng tối ưu trên thiết bị di động, chạy tại cổng \texttt{3000}.
        \item \textbf{API Application (Backend Service):} Ứng dụng NestJS 12 (TypeScript) cung cấp toàn bộ RESTful API, đóng gói trong Docker Container chạy tại cổng \texttt{3001}.
        \item \textbf{Database (Cơ sở dữ liệu quan hệ):} PostgreSQL 16 đóng gói trong Docker Container chạy tại cổng \texttt{5432}, gắn ổ đĩa lưu trữ bền vững (Persistent Storage).
    \end{itemize}

    \item \textbf{C4 Level 3 -- Component (Phân rã Backend NestJS):}
    Ứng dụng Backend được tổ chức theo kiến trúc \textbf{Modular Monolith} với các Module nghiệp vụ khép kín:
    \begin{itemize}[leftmargin=*,noitemsep]
        \item \texttt{AuthModule}: Phụ trách xác thực người dùng, băm mật khẩu \texttt{bcrypt}, cấp phát và kiểm tra JWT token qua \texttt{JwtAuthGuard}.
        \item \texttt{ProductsModule} \& \texttt{ToppingsModule}: Phụ trách truy vấn thực đơn, danh mục và cập nhật cờ bật/tắt món.
        \item \texttt{OrdersModule}: Trọng tâm nghiệp vụ, quản lý máy trạng thái đơn hàng (FSM 7 bước), kiểm tra tồn kho nguyên liệu và thực thi khóa lạc quan.
        \item \texttt{PaymentsModule}: Quản lý giao dịch thanh toán không tiền mặt và kiểm soát chống trùng lặp qua bảng băm \texttt{idempotencyKey}.
        \item \texttt{UsersModule}: Quản lý thông tin tài khoản, tích lũy điểm thưởng và phân hạng thành viên.
        \item \texttt{PrismaModule}: Dịch vụ dùng chung cung cấp kết nối cơ sở dữ liệu PostgreSQL an toàn kiểu (Type-safe).
    \end{itemize}
\end{enumerate}

\subsubsection{Phân tích lý do lựa chọn, ưu điểm, hạn chế và sự đánh đổi (Trade-offs)}

\begin{itemize}[leftmargin=*]
    \item \textbf{Lý do chọn Modular Monolith thay vì Microservices:} Hệ thống BrewLite phục vụ quy mô chuỗi cà phê vừa và nhỏ với lưu lượng hàng nghìn đơn mỗi ngày. Việc lựa chọn Modular Monolith giúp toàn bộ giao dịch đặt hàng, khóa giữ kho và thanh toán được thực thi trong một Database Transaction (ACID) duy nhất của PostgreSQL, loại bỏ hoàn toàn bài toán phân tán dữ liệu phức tạp (như Two-Phase Commit hay Saga Pattern) vốn dễ gây rủi ro sai lệch dữ liệu tài chính trong microservices.
    \item \textbf{Đánh đổi (Trade-offs):} Toàn bộ backend được triển khai trong cùng một tiến trình (Single Deployment Unit). Nếu một module gặp sự cố ngốn CPU, toàn bộ API có thể bị ảnh hưởng. Tuy nhiên, bù lại nhóm đạt được tốc độ phát triển cực nhanh, chi phí vận hành hạ tầng Cloud ở mức tối thiểu và độ tin cậy dữ liệu đạt 100\%.
\end{itemize}

\subsubsection{Tiêu chuẩn Định nghĩa Hoàn thành (Definition of Done -- DoD)}
Một User Story chỉ được chuyển sang cột \textbf{Done} trên bảng quản lý khi thỏa mãn cả 6 tiêu chuẩn sau:
\begin{enumerate}[leftmargin=*,noitemsep]
    \item \textbf{Chức năng (Functionality):} Đạt 100\% các tiêu chí Acceptance Criteria quy định trong SRS.
    \item \textbf{Kiểm thử (Testing):} Đã viết Unit Test / Integration Test bao phủ luồng chính và biên; pass 100\%.

\subsubsection{Đặc tả 7 yêu cầu kỹ thuật trong Pipeline CD}
\begin{enumerate}[leftmargin=*]
    \item \textbf{Quality Gate nghiêm ngặt:} Trước khi cho phép đóng gói, pipeline tự động chạy kiểm tra linting (\texttt{oxlint}), khởi tạo container PostgreSQL dịch vụ, chạy Prisma migration, nạp test seed và thực thi toàn bộ bài kiểm thử tự động (\texttt{npm test}). Bất kỳ bước nào thất bại, pipeline dừng ngay lập tức.
    \item \textbf{Tạo Docker Image gắn tag phiên bản:} Docker Image được gắn nhãn định danh kép: commit Git SHA (để đối soát mã nguồn) và Semantic Version (ví dụ: \texttt{v1.0.0}).
    \item \textbf{Đẩy Image lên Container Registry:} Đẩy Docker Image đã đóng gói lên GitHub Container Registry (GHCR) hoặc Docker Hub an toàn.
    \item \textbf{Triển khai vào Production bằng Secret:} Sử dụng Webhook bí mật \texttt{RENDER\_DEPLOY\_HOOK} lưu trong GitHub Secrets để kích hoạt Render kéo image mới mà không để lộ mật khẩu.
    \item \textbf{Chạy Database Migration an toàn:} Tuyệt đối không dùng lệnh \texttt{prisma db push} có nguy cơ mất dữ liệu. Hệ thống sử dụng lệnh \texttt{npx prisma migrate deploy} trong chỉ lệnh \texttt{CMD} của Dockerfile để bảo đảm tính toàn vẹn dữ liệu.
    \item \textbf{Smoke Test tự động sau triển khai:} Sau khi kích hoạt deploy, pipeline chờ container khởi động (grace period 45s) và thực hiện kiểm tra tự động (retry 6 lần $\times$ 15s) tới endpoint \texttt{GET /health} trên môi trường Production.
    \item \textbf{Cơ chế Rollback khi triển khai thất bại:}
    \begin{itemize}[leftmargin=*,noitemsep]
        \item \textit{Tầng 1 (Render Zero-Downtime):} Render chỉ chuyển đổi lưu lượng truy cập khi container mới pass Health Check. Nếu container mới bị lỗi crash loop, Render giữ nguyên container phiên bản cũ đang chạy.
        \item \textit{Tầng 2 (REST API Rollback):} Pipeline CD tự động phát lệnh đảo ngược qua Render REST API nếu Smoke Test thất bại.
        \item \textit{Tầng 3 (SOP Rollback thủ công):} Đội ngũ kỹ sư có thể rollback nhanh trong 30 giây trực tiếp từ Render Dashboard tại mục Web Service $\rightarrow$ Deploys $\rightarrow$ Rollback to this deploy.
    \end{itemize}
\end{enumerate}
